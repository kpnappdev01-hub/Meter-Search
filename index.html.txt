<script>
  // ================= 1. พื้นฐาน =================
  const IS_GAS = (typeof google !== 'undefined' && google.script && google.script.run);
  const SESSION_KEY = 'pea_lookup_session';   // แยกจากระบบของช่างคนละคีย์
  const DEVICE_KEY  = 'pea_lookup_device';    // ใช้นับการกรอก PIN ผิดแยกตามเครื่อง
  const MIN_QUERY_LEN = 3;

  let session = null;
  let lastRes = null;       // ผลค้นหาล่าสุด เก็บไว้ให้ปุ่ม "กลับ" แสดงซ้ำได้โดยไม่ต้องค้นใหม่
  let lastResults = [];
  let current = null;       // งานที่เปิดการ์ดรายละเอียดอยู่
  const photoCache = {};    // key -> { thumb, full }  (ใช้ key ไม่ใช้เลขแถว เพราะแถวเลื่อนได้)

  function $(id) { return document.getElementById(id); }
  function show(id) { $(id).classList.remove('hide'); }
  function hide(id) { $(id).classList.add('hide'); }

  function escapeHtml(str) {
    return String(str === null || str === undefined ? '' : str)
      .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
      .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
  }

  function callServer(fnName) {
    const args = Array.prototype.slice.call(arguments, 1);
    return new Promise(function (resolve, reject) {
      if (!IS_GAS) { reject(new Error('หน้านี้ยังไม่ได้เปิดผ่าน Google Apps Script')); return; }
      google.script.run
        .withSuccessHandler(resolve)
        .withFailureHandler(function (err) { reject(new Error(err && err.message ? err.message : String(err))); })
        [fnName].apply(null, args);
    });
  }

  function callAuth(fnName) {
    const args = Array.prototype.slice.call(arguments, 1);
    return callServer.apply(null, [fnName, session ? session.token : ''].concat(args))
      .then(function (res) {
        if (res && res.authError) {
          clearSession();
          showLogin('หมดเวลาใช้งาน กรุณาเข้าสู่ระบบใหม่');
          throw new Error('SESSION_EXPIRED');
        }
        if (!res || !res.success) throw new Error(res && res.message ? res.message : 'ไม่ทราบสาเหตุ');
        return res;
      });
  }

  function store(key, value) { try { localStorage.setItem(key, JSON.stringify(value)); return true; } catch (e) { return false; } }
  function load(key) { try { const v = localStorage.getItem(key); return v ? JSON.parse(v) : null; } catch (e) { return null; } }
  function drop(key) { try { localStorage.removeItem(key); } catch (e) {} }

  function deviceId() {
    let id = load(DEVICE_KEY);
    if (id) return id;
    try {
      const a = new Uint8Array(16);
      crypto.getRandomValues(a);
      id = Array.prototype.map.call(a, function (b) { return ('0' + b.toString(16)).slice(-2); }).join('');
    } catch (e) {
      id = Date.now().toString(16) + Math.random().toString(16).slice(2);
    }
    store(DEVICE_KEY, id);
    return id;
  }

  function showLoading(text) { $('loadingText').textContent = text || 'กำลังทำงาน...'; show('loading'); }
  function hideLoading() { hide('loading'); }

  function showMessage(title, desc, icon) {
    $('msgTitle').textContent = title;
    $('msgDesc').textContent = desc || '';
    $('msgIcon').textContent = icon || '';
    show('msgBox');
  }
  function hideMessage() { hide('msgBox'); }

  // ================= 2. เข้าสู่ระบบ =================
  window.onload = function () {
    if (!IS_GAS) { showLogin('หน้านี้ยังไม่ได้เปิดผ่าน Google Apps Script จึงเข้าสู่ระบบไม่ได้'); return; }

    $('loginPin').addEventListener('keyup', function (e) { if (e.key === 'Enter') doLogin(); });
    $('q').addEventListener('keyup', function (e) { if (e.key === 'Enter') doSearch(); });

    const saved = load(SESSION_KEY);
    if (!saved || !saved.token) { showLogin(); return; }

    showLoading('กำลังตรวจสอบสิทธิ์...');
    callServer('checkSession', saved.token)
      .then(function (res) {
        hideLoading();
        if (res && res.valid) {
          session = { token: saved.token, code: res.code, name: res.name };
          startApp();
        } else { clearSession(); showLogin(); }
      })
      .catch(function () { hideLoading(); showLogin(); });
  };

  function showLogin(errorText) {
    hide('appScreen');
    show('loginScreen');
    const err = $('loginError');
    if (errorText) { err.textContent = errorText; err.classList.remove('hide'); }
    else err.classList.add('hide');
    $('loginPin').value = '';
  }

  function clearSession() { session = null; drop(SESSION_KEY); }

  function doLogin() {
    const code = $('loginCode').value.trim();
    const pin  = $('loginPin').value.trim();
    const err  = $('loginError');
    const btn  = $('loginBtn');

    if (!code || !pin) {
      err.textContent = 'กรุณากรอกรหัสผู้ใช้และ PIN';
      err.classList.remove('hide');
      return;
    }

    err.classList.add('hide');
    btn.disabled = true;
    btn.innerHTML = '<span class="spin"></span> กำลังตรวจสอบ...';
    const restore = function () { btn.disabled = false; btn.textContent = 'เข้าสู่ระบบ'; };

    callServer('login', code, pin, deviceId())
      .then(function (res) {
        restore();
        if (!res || !res.success) {
          err.textContent = res ? res.message : 'เข้าสู่ระบบไม่สำเร็จ';
          err.classList.remove('hide');
          $('loginPin').value = '';
          return;
        }
        session = { token: res.token, code: res.code, name: res.name };
        store(SESSION_KEY, session);
        $('loginPin').value = '';
        startApp();
      })
      .catch(function (e) {
        restore();
        err.textContent = e.message;
        err.classList.remove('hide');
      });
  }

  function doLogout() {
    const token = session ? session.token : '';
    clearSession();
    lastRes = null;
    lastResults = [];
    current = null;
    $('results').innerHTML = '';
    $('detail').innerHTML = '';
    $('q').value = '';
    showLogin();
    if (IS_GAS && token) callServer('logout', token).catch(function () {});
  }

  function startApp() {
    hide('loginScreen');
    show('appScreen');
    $('userLabel').textContent = session.name + ' (' + session.code + ')';
    setTimeout(function () { $('q').focus(); }, 200);
  }

  // ================= 3. ค้นหา =================
  function doSearch() {
    const q = $('q').value.trim();
    if (q.length < MIN_QUERY_LEN) {
      showMessage('พิมพ์สั้นเกินไป', 'กรุณาพิมพ์อย่างน้อย ' + MIN_QUERY_LEN + ' ตัวอักษร', '⌨️');
      return;
    }

    const btn = $('searchBtn');
    btn.disabled = true;
    btn.innerHTML = '<span class="spin"></span>';
    $('detail').innerHTML = '';
    $('results').innerHTML = '';
    current = null;

    callAuth('searchMeters', q)
      .then(function (res) {
        btn.disabled = false;
        btn.textContent = 'ค้นหา';
        lastRes = res;
        lastResults = res.results || [];

        if (!lastResults.length) {
          $('results').innerHTML =
            '<div class="card"><div class="muted">ไม่พบข้อมูลที่ตรงกับ "' + escapeHtml(res.query) + '"<br>' +
            'ลองพิมพ์เลขให้ครบขึ้น หรือค้นด้วยชื่อลูกค้าแทน</div></div>';
          return;
        }

        if (lastResults.length === 1) { renderDetail(lastResults[0]); return; }
        renderResults(res);
      })
      .catch(function (err) {
        btn.disabled = false;
        btn.textContent = 'ค้นหา';
        if (err.message === 'SESSION_EXPIRED') return;
        showMessage('ค้นหาไม่สำเร็จ', err.message, '⚠️');
      });
  }

  function renderResults(res) {
    let html = '<p class="hint" style="margin:2px 2px 8px">พบ ' + res.total + ' รายการ' +
               (res.truncated ? ' (แสดง ' + res.limit + ' รายการแรก กรุณาพิมพ์ให้เจาะจงขึ้น)' : '') + '</p>';

    html += lastResults.map(function (t, i) {
      return '<button class="result" onclick="openResult(' + i + ')">' +
             '<div class="pea mono">' + escapeHtml(t.peaNo || '-') + '</div>' +
             '<div class="who">' + escapeHtml(t.customer) + '</div>' +
             (t.address ? '<div class="sub">' + escapeHtml(t.address) + '</div>' : '') +
             '<div class="sub" style="margin-top:6px">' +
               (t.hasPhoto ? '<span class="tag tag-ok">มีรูปติดตั้ง</span> ' : '<span class="tag tag-wait">ยังไม่ได้บันทึกงาน</span> ') +
               '<span class="tag tag-hit">ตรงที่ ' + escapeHtml(t.matchedBy.join(', ')) + '</span>' +
             '</div>' +
           '</button>';
    }).join('');

    $('results').innerHTML = html;
  }

  function openResult(i) {
    const t = lastResults[i];
    if (!t) return;
    $('results').innerHTML = '<button class="btn btn-ghost btn-sm" style="margin-bottom:10px" onclick="backToResults()">← กลับไปที่ผลการค้นหา</button>';
    renderDetail(t);
  }

  function backToResults() {
    if (!lastRes) return;
    current = null;
    $('detail').innerHTML = '';
    renderResults(lastRes);
  }

  // ================= 4. การ์ดรายละเอียด =================
  function mapLink(coords) { return 'https://maps.google.com/?q=' + encodeURIComponent(coords); }

  function kv(label, value) {
    const v = String(value === null || value === undefined ? '' : value).trim();
    return '<div class="kv"><div class="k">' + escapeHtml(label) + '</div>' +
           '<div class="v ' + (v ? '' : 'empty') + '">' + (v ? escapeHtml(v) : '-') + '</div></div>';
  }

  function renderDetail(t) {
    current = t;

    let html = '<div class="card">';

    html += '<div class="detail-head">' +
              (t.reqType ? '<span class="tag tag-req">' + escapeHtml(t.reqType) + '</span>' : '') +
              '<div class="pea mono" style="margin-top:6px">' + escapeHtml(t.peaNo || '-') + '</div>' +
              '<div class="who">' + escapeHtml(t.customer) + '</div>' +
              '<div class="addr">' + escapeHtml(t.address || 'ไม่ระบุที่อยู่') + '</div>' +
            '</div>';

    html += kv('หมายเลขจดหน่วย', t.meterNo) + kv('PEA.เดิม', t.peaOld);

    // พิกัดตามคำร้อง
    html += '<div class="geo req">' +
              '<h3>📍 พิกัดตามคำร้อง</h3>';
    if (t.coords) {
      html += '<div class="val mono">' + escapeHtml(t.coords) + '</div>' +
              '<a class="maplink" target="_blank" rel="noopener" href="' + mapLink(t.coords) + '">เปิดแผนที่</a>';
    } else {
      html += '<div class="meta">ไม่มีพิกัดในคำร้อง</div>';
    }
    html += '</div>';

    // พิกัดหน้างานจริง
    if (t.siteCoords) {
      const meta = [t.coordSource, t.coordAccuracy ? '± ' + t.coordAccuracy + ' ม.' : '']
                     .filter(Boolean).join(' · ');
      html += '<div class="geo site">' +
                '<h3>🎯 พิกัดหน้างานจริง</h3>' +
                '<div class="val mono">' + escapeHtml(t.siteCoords) + '</div>' +
                (meta ? '<div class="meta">' + escapeHtml(meta) + '</div>' : '') +
                '<a class="maplink" target="_blank" rel="noopener" href="' + mapLink(t.siteCoords) + '">เปิดแผนที่</a>' +
              '</div>';
    } else {
      html += '<div class="geo wait"><h3>🎯 พิกัดหน้างานจริง</h3>' +
              '<div class="meta">ช่างยังไม่ได้บันทึกงานนี้ — ใช้พิกัดตามคำร้องไปก่อน</div></div>';
    }

    // รูปติดตั้งเสร็จ
    if (t.hasPhoto) {
      html += '<div class="photo-box" id="photoBox" onclick="openPhoto()">' +
                '<div class="spin spin-lg spin-dark" id="thumbSpin"></div>' +
              '</div>';
    }

    html += '</div>';
    $('detail').innerHTML = html;
    window.scrollTo({ top: 0, behavior: 'smooth' });

    callAuth('logView', t.peaNo).catch(function () {});
    if (t.hasPhoto) loadThumb(t);
  }

  // ================= 5. รูปภาพ =================
  function loadThumb(t) {
    const box = $('photoBox');
    if (!box) return;

    const paint = function (src) {
      box.innerHTML = '<img src="' + src + '" alt="รูปติดตั้งเสร็จ">' +
                      '<div class="cap">รูปติดตั้งเสร็จ — แตะเพื่อดูเต็มจอ</div>';
    };

    const cached = photoCache[t.key];
    if (cached && cached.thumb) { paint(cached.thumb); return; }

    callAuth('getInstallPhoto', t.row, false, t.key)
      .then(function (res) {
        photoCache[t.key] = photoCache[t.key] || {};
        photoCache[t.key].thumb = res.dataUrl;
        if (current === t) paint(res.dataUrl);
      })
      .catch(function (err) {
        if (err.message === 'SESSION_EXPIRED') return;
        if (current === t) box.innerHTML = '<div class="muted">แสดงรูปไม่ได้: ' + escapeHtml(err.message) + '</div>';
      });
  }

  function openPhoto() {
    const t = current;
    if (!t) return;

    show('photoModal');
    const img = $('photoFull');
    const cached = photoCache[t.key];

    if (cached && cached.full) { img.src = cached.full; hide('photoSpin'); return; }

    // ระหว่างรอไฟล์เต็ม แสดงรูปย่อไปพลางก่อน จะได้ไม่เห็นจอดำเปล่า ๆ
    img.src = (cached && cached.thumb) ? cached.thumb : '';
    show('photoSpin');

    callAuth('getInstallPhoto', t.row, true, t.key)
      .then(function (res) {
        hide('photoSpin');
        photoCache[t.key] = photoCache[t.key] || {};
        photoCache[t.key].full = res.dataUrl;
        if (current === t) img.src = res.dataUrl;
      })
      .catch(function (err) {
        hide('photoSpin');
        if (err.message === 'SESSION_EXPIRED') { closePhoto(); return; }
        closePhoto();
        showMessage('เปิดรูปไม่ได้', err.message, '🖼️');
      });
  }

  function closePhoto() { hide('photoModal'); }

  document.addEventListener('keydown', function (e) {
    if (e.key === 'Escape' && !$('photoModal').classList.contains('hide')) closePhoto();
  });
